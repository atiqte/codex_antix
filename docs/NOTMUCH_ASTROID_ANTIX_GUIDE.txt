# notmuch + Astroid Search Layer for antiX Maildir++

Validated on 2026-07-05 for antiX Linux runit with zzzFM/IceWM.

The reliable stack is:

- notmuch for fast search.
- The read-only localhost browser viewer for search/view.
- Evolution Flatpak for normal account-based reply/send.
- Astroid installed and configured, but experimental on this VM because message bodies stayed blank.

Data safety overrides speed. The 59G archive is intentionally not indexed yet because a later audit showed count drift: `/mail/Mailstore/evolution/local-maildir` showed 48,564 `cur` files while the earlier verified restore ledger said 48,720.

## Validated layout

- notmuch config: `~/.config/notmuch/default/config`
- notmuch database: `/mail/SearchIndex/notmuch/default`
- notmuch mail root: `/mail/Mailstore`
- Pilot scope: `/mail/Mailstore/mbsync/provider-live` plus `/mail/Mailstore/evolution/betterbird-delta-maildirpp-20260704`
- Pilot count: 444 unique messages, 846 files
- Safety flags: `maildir.synchronize_flags=false`, `index.decrypt=false`
- Backups: `/mail/Backups/notmuch`
- Browser viewer: `http://127.0.0.1:8765/`

## Operating rules

1. Never delete or rebuild Maildir data to fix notmuch.
2. The notmuch database is disposable; the mail files are not.
3. Pause mbsync before full reindex work, then resume it.
4. Back up tags before reindex or scope changes.
5. Keep `maildir.synchronize_flags=false`.
6. Do not use `xdg-open` on raw Maildir files for Evolution. Validation showed Evolution opens the wrong Berkeley mbox import wizard.
7. Use Evolution only from its configured Maildir accounts for reply, forward, edit, and send.

def search_form(q):
    return f"""
<form class="search" method="get" action="/">
  <input type="text" name="q" value="{html.escape(q, quote=True)}" autofocus>
  <button type="submit">Search</button>

<div class="small code">Examples: tag:inbox | from:tagindustries.com.sg | date:2026..2026 and tag:attachment | id:&lt;message-id&gt;</div>

def parse_search_line(line):
    # notmuch text format starts with thread:<id> then human summary.
    if not line.startswith("thread:"):
        return None, line
    first, rest = (line.split(" ", 1) + [""])[:2]
    return first.replace("thread:", "", 1), rest.strip()

def render_search(q):
    count = run_notmuch(["count", q])
    file_count = run_notmuch(["count", "--output=files", q])
    res = run_notmuch(["search", "--limit", str(MAX_RESULTS), q])
    body = search_form(q)
    if count.returncode != 0:
        body += f"<div class='notice bad'><strong>notmuch error</strong><pre>{html.escape(count.stderr or res.stderr)}</pre></div>"
        return page("notmuch search error", body)
    body += f"<div class='notice ok'>Unique messages: <strong>{html.escape(count.stdout.strip())}</strong>; files: <strong>{html.escape(file_count.stdout.strip())}</strong>; showing first {MAX_RESULTS} threads.</div>"
    if res.returncode != 0:
        body += f"<div class='notice bad'><pre>{html.escape(res.stderr)}</pre></div>"
        return page("notmuch search error", body)
    rows = []
    for line in res.stdout.splitlines():
        tid, rest = parse_search_line(line)
        if tid:
            url = "/thread?" + urllib.parse.urlencode({"id": tid})
            rows.append(f"<tr><td class='code'><a href='{url}'>{html.escape(tid)}</a></td><td>{html.escape(rest)}</td></tr>")
        else:
            rows.append(f"<tr><td></td><td>{html.escape(line)}</td></tr>")
    if not rows:
        body += "<div class='notice'>No results.</div>"
    else:
        body += "<table><thead><tr><th>Thread</th><th>Summary</th></tr></thead><tbody>" + "\n".join(rows) + "</tbody></table>"
    return page("notmuch search", body)

def iter_body_parts(msg):
    if msg.is_multipart():
        body = msg.get_body(preferencelist=("html", "plain"))
        if body is not None:
            yield body
        else:
            for part in msg.walk():
                if part.get_content_maintype() == "text":
                    yield part
    else:
        yield msg

def body_html_for_message(msg):
    parts = list(iter_body_parts(msg))
    if not parts:
        return "<div class='notice'>No text/html or text/plain body part found.</div>"
    part = parts[0]
    ctype = part.get_content_type()
    try:
        content = part.get_content()
    except Exception as e:
        return f"<div class='notice bad'>Could not decode body: {html.escape(str(e))}</div>"
    if ctype == "text/html":
        src = html.escape(content, quote=True)
        return f"<div class='small'>HTML part shown in a sandboxed iframe; remote network/script execution is blocked by browser policy.</div><iframe sandbox srcdoc=\"{src}\"></iframe>"
    return f"<pre>{html.escape(str(content))}</pre>"

def render_message_file(path):
    p = Path(path)
    try:
        with p.open("rb") as f:
            msg = BytesParser(policy=policy.default).parse(f)
    except Exception as e:
        return f"<div class='message'><div class='meta bad'>Could not parse {html.escape(path)}: {html.escape(str(e))}</div></div>"

    headers = []
    for name in ("Subject", "From", "To", "Cc", "Date", "Message-ID"):
        val = msg.get(name, "")
        if val:
            headers.append(f"<div><strong>{html.escape(name)}:</strong> {html.escape(str(val))}</div>")

    attachments = []
    for part in msg.walk():
        filename = part.get_filename()
        disp = part.get_content_disposition()
        ctype = part.get_content_type()
        if filename or disp == "attachment":
            attachments.append(html.escape(filename or ctype))
    attach_html = ""
    if attachments:
        attach_html = "<div class='attach'><strong>Attachments:</strong> " + ", ".join(attachments) + "</div>"

    return f"""
<section class="message">
  <div class="meta">
    {''.join(headers)}
    <div class="path"><strong>File:</strong> {html.escape(path)}</div>
    {attach_html}
  </div>
  <div class="body">{body_html_for_message(msg)}</div>
</section>
"""

def render_thread(tid):
    q = f"thread:{tid}"
    files = run_notmuch(["search", "--output=files", q])
    summary = run_notmuch(["search", "--limit", "1", q])
    body = f"<p><a class='button' href='/'>Back to search</a></p>"
    body += f"<h2>Thread {html.escape(tid)}</h2>"
    if summary.returncode == 0:
        body += f"<pre>{html.escape(summary.stdout.strip())}</pre>"
    if files.returncode != 0:
        body += f"<div class='notice bad'><pre>{html.escape(files.stderr)}</pre></div>"
        return page("thread error", body)

    paths = []
    seen = set()
    for line in files.stdout.splitlines():
        if line and line not in seen:
            seen.add(line)
            paths.append(line)
    body += f"<div class='notice'>Message files in this thread: {len(paths)}. Showing first {MAX_THREAD_FILES}.</div>"
    for path in paths[:MAX_THREAD_FILES]:
        body += render_message_file(path)
    return page("thread " + tid, body)

def render_status():
    c = run_notmuch(["count", "*"])
    f = run_notmuch(["count", "--output=files", "*"])
    tags = run_notmuch(["search", "--output=tags", "*"])
    body = "<h2>Status</h2>"

    body += "<h3>Tags</h3><pre>" + html.escape(tags.stdout) + "</pre>"
    body += "<h3>Data Safety</h3><ul><li>Read-only viewer.</li><li>No tag edits.</li><li>No mail file writes.</li><li>Use Evolution for reply/send.</li></ul>"
    return page("status", body)

class Handler(BaseHTTPRequestHandler):
    def log_message(self, fmt, *args):
        sys.stderr.write("[%s] %s\n" % (self.log_date_time_string(), fmt % args))

    def do_GET(self):
        parsed = urllib.parse.urlparse(self.path)
        qs = urllib.parse.parse_qs(parsed.query)
        try:
            if parsed.path == "/":
                q = qs.get("q", [DEFAULT_QUERY])[0].strip() or DEFAULT_QUERY
                content = render_search(q)
            elif parsed.path == "/thread":
                tid = qs.get("id", [""])[0].strip()
                if not tid:
                    content = page("missing thread", "<div class='notice bad'>Missing thread id.</div>")
                else:
                    content = render_thread(tid)
            elif parsed.path == "/status":
                content = render_status()
            else:
                self.send_response(404)
                self.end_headers()
                self.wfile.write(b"not found")
                return
            data = content.encode("utf-8", "replace")
            self.send_response(200)
            self.send_header("Content-Type", "text/html; charset=utf-8")
            self.send_header("Content-Length", str(len(data)))
            self.end_headers()
            self.wfile.write(data)
        except Exception as e:
            data = page("error", "<div class='notice bad'><pre>" + html.escape(repr(e)) + "</pre></div>").encode()
            self.send_response(500)
            self.send_header("Content-Type", "text/html; charset=utf-8")
            self.send_header("Content-Length", str(len(data)))
            self.end_headers()
            self.wfile.write(data)

def pick_port(start):
    for port in range(start, start + 20):
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            try:
                s.bind(("127.0.0.1", port))
            except OSError:
                continue
            return port
    raise SystemExit("no free localhost port found")

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--port", type=int, default=8765)
    ap.add_argument("--no-open", action="store_true")
    args = ap.parse_args()

    if not CONFIG.exists():
        raise SystemExit(f"missing notmuch config: {CONFIG}")

    port = pick_port(args.port)
    server = ThreadingHTTPServer(("127.0.0.1", port), Handler)
    url = f"http://127.0.0.1:{port}/"
    print(f"notmuch_browser_url={url}", flush=True)
    print("read_only=yes", flush=True)
    if not args.no_open:
        threading.Thread(target=lambda: (time.sleep(0.8), webbrowser.open(url)), daemon=True).start()
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        pass

if __name__ == "__main__":
