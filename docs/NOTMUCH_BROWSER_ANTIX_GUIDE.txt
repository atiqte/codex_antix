# notmuch Browser-Only Search Layer for antiX Maildir++

Historical pilot guide. For a new VM and the complete current Go browser, use
`docs/NOTMUCH_GO_BROWSER_SERVICE_GUIDE.html` or `.txt`. The archive exclusion
and small pilot counts below record the older current-VM pilot; they are not the
approved fresh-VM policy. Fresh VMs restore the verified 48,720-file package and
index all of `local-maildir`.

Validated on 2026-07-05 for antiX Linux runit with zzzFM/IceWM.

Current result:

- Astroid was purged and its user settings were backed up before removal.
- notmuch remains the fast search backend.
- `notmuch-browser-pilot` is now the primary GUI viewer.
- The browser viewer is single-email mode, not threaded mode.
- Evolution Flatpak remains the normal reply/forward/send client through configured Maildir accounts.
- The 59G archive remains excluded until count drift is reviewed.

Validated final pilot state:

- notmuch config: `~/.config/notmuch/default/config`
- database: `/mail/SearchIndex/notmuch/default`
- mail root: `/mail/Mailstore`
- safety flags: `maildir.synchronize_flags=false`, `index.decrypt=false`
- indexed files after refresh: 867
- unique messages after refresh: 457
- indexed provider-live files: 620
- indexed restored delta files: 247
- indexed 59G archive files: 0
- browser URL: `http://127.0.0.1:8765/`
- viewer mode: `single_email`

Data safety rules:

1. Never delete or rebuild Maildir data to fix notmuch.
2. The notmuch database is disposable; the mail files are not.
3. Back up tags before reindex or scope changes.
4. Keep `maildir.synchronize_flags=false`.
5. Use browser viewer for search/view only.
6. Use Evolution for reply/forward/edit/send.
7. Do not use Evolution as a raw Maildir file opener from notmuch.

def nm(args, timeout=90):
    return subprocess.run(["notmuch", "--config", str(CONFIG)] + args, text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=timeout)

def page(title, body):
    return f"""<!doctype html><html><head><meta charset='utf-8'>
<meta http-equiv="Content-Security-Policy" content="default-src 'self'; img-src 'self' data:; style-src 'unsafe-inline'; frame-src 'self' about:; script-src 'none'; object-src 'none'">
<title>{html.escape(title)}</title><style>{CSS}</style></head><body>
<header><strong>notmuch browser pilot</strong><a href='/'>Search</a><a href='/status'>Status</a><span class='small'>single-email read-only viewer; use Evolution for reply/send</span></header>
<main>{body}</main></body></html>"""

def form(q):
    return f"""<form class='search' method='get' action='/'><input type='text' name='q' value='{html.escape(q, quote=True)}' autofocus><button>Search</button></form>
<div class='small code'>Examples: tag:inbox | from:tagindustries.com.sg | date:2026..2026 and tag:attachment | id:&lt;message-id&gt;</div>"""

def mid_line(line):
    line = line.strip()
    return line[3:] if line.startswith("id:") else line

def files_for(mid):
    r = nm(["search", "--output=files", "id:" + mid])
    if r.returncode != 0:
        return [], r.stderr.strip()
    seen, out = set(), []
    for line in r.stdout.splitlines():
        if line and line not in seen:
            seen.add(line); out.append(line)
    return out, ""

def tags_for(mid):
    r = nm(["search", "--output=tags", "id:" + mid])
    return [x.strip() for x in r.stdout.splitlines() if x.strip()] if r.returncode == 0 else []

def parse_file(path):
    with Path(path).open("rb") as fh:
        return BytesParser(policy=policy.default).parse(fh)

def hv(msg, name):
    v = msg.get(name, "")
    return str(v) if v else ""

def summary(mid):
    files, err = files_for(mid)
    if not files:
        return {"id": mid, "date": "", "from": "", "subject": "(file missing)", "tags": [], "file_count": 0, "error": err or "notmuch returned no files"}
    try:
        msg = parse_file(files[0])
        return {"id": mid, "date": hv(msg, "Date"), "from": hv(msg, "From"), "subject": hv(msg, "Subject") or "(no subject)", "tags": tags_for(mid), "file_count": len(files), "error": ""}
    except Exception as e:
        return {"id": mid, "date": "", "from": "", "subject": "(parse error)", "tags": tags_for(mid), "file_count": len(files), "error": str(e)}

def render_search(q):
    c = nm(["count", q]); f = nm(["count", "--output=files", q]); ids = nm(["search", "--output=messages", "--limit", str(MAX_RESULTS), q])
    body = form(q)
    if c.returncode or ids.returncode:
        return page("notmuch search error", body + "<div class='notice bad'><pre>" + html.escape(c.stderr or ids.stderr) + "</pre></div>")
    mids = [mid_line(x) for x in ids.stdout.splitlines() if x.strip()]
    body += f"<div class='notice ok'>Unique messages: <strong>{html.escape(c.stdout.strip())}</strong>; files: <strong>{html.escape(f.stdout.strip())}</strong>; showing first {MAX_RESULTS} individual messages.</div>"
    if not mids:
        return page("notmuch search", body + "<div class='notice'>No results.</div>")
    rows = []
    for mid in mids:
        s = summary(mid)
        url = "/message?" + urllib.parse.urlencode({"id": mid})
        tag_html = "".join(f"<span class='tag'>{html.escape(t)}</span>" for t in s["tags"])
        err = f"<div class='small bad'>Error: {html.escape(s['error'])}</div>" if s["error"] else ""
        rows.append(f"<tr><td class='small'>{html.escape(s['date'])}</td><td>{html.escape(s['from'])}</td><td><a href='{url}'>{html.escape(s['subject'])}</a>{err}<div class='path code'>{html.escape(mid)}</div></td><td>{tag_html}</td><td class='code'>{s['file_count']}</td></tr>")
    return page("notmuch search", body + "<table><thead><tr><th>Date</th><th>From</th><th>Subject / Message-ID</th><th>Tags</th><th>Files</th></tr></thead><tbody>" + "\n".join(rows) + "</tbody></table>")

def body_part(msg):
    part = msg.get_body(preferencelist=("html", "plain")) if msg.is_multipart() else msg
    if part is None:
        return "<div class='notice'>No text/html or text/plain body part found.</div>"
    try:
        content = part.get_content()
    except Exception as e:
        return f"<div class='notice bad'>Could not decode body: {html.escape(str(e))}</div>"
    if part.get_content_type() == "text/html":
        return "<div class='small'>HTML part shown in a sandboxed iframe; remote network/script execution is blocked.</div><iframe sandbox srcdoc=\"" + html.escape(str(content), quote=True) + "\"></iframe>"
    return "<pre>" + html.escape(str(content)) + "</pre>"

def render_message(mid, file_index):
    files, err = files_for(mid)
    body = "<p><a class='button' href='/'>Back to search</a></p><h2>Message</h2><pre>" + html.escape("id:" + mid) + "</pre>"
    if err:
        return page("message error", body + "<div class='notice bad'><pre>" + html.escape(err) + "</pre></div>")
    if not files:
        return page("message missing", body + "<div class='notice bad'>No files found for this Message-ID.</div>")
    try:
        idx = int(file_index or "0")
    except ValueError:
        idx = 0
    idx = idx if 0 <= idx < len(files) else 0
    if len(files) > 1:
        choices = []
        for i, path in enumerate(files[:40]):
            url = "/message?" + urllib.parse.urlencode({"id": mid, "file": str(i)})
            label = f"{i + 1}: {path}"
            choices.append("<strong>" + html.escape(label) + "</strong>" if i == idx else f"<a href='{url}'>{html.escape(label)}</a>")
        body += "<div class='notice'>Duplicate/copy files for this Message-ID: " + str(len(files)) + ". Showing selected file only.<br>" + "<br>".join(choices) + "</div>"
    msg = parse_file(files[idx])
    headers = "".join(f"<div><strong>{name}:</strong> {html.escape(hv(msg, name))}</div>" for name in ("Subject", "From", "To", "Cc", "Date", "Message-ID") if hv(msg, name))
    attaches = [html.escape(p.get_filename() or p.get_content_type()) for p in msg.walk() if p.get_filename() or p.get_content_disposition() == "attachment"]
    attach_html = "<div><strong>Attachments:</strong> " + ", ".join(attaches) + "</div>" if attaches else ""
    tag_html = "".join(f"<span class='tag'>{html.escape(t)}</span>" for t in tags_for(mid))
    body += f"<section class='message'><div class='meta'>{headers}<div><strong>Tags:</strong> {tag_html}</div><div class='path'><strong>Selected file:</strong> {html.escape(files[idx])}</div>{attach_html}</div><div class='body'>{body_part(msg)}</div></section>"
    return page("message", body)

class Handler(BaseHTTPRequestHandler):
    def log_message(self, fmt, *args):
        sys.stderr.write("[%s] %s\n" % (self.log_date_time_string(), fmt % args))
    def do_GET(self):
        parsed = urllib.parse.urlparse(self.path); qs = urllib.parse.parse_qs(parsed.query)
        try:
            if parsed.path == "/":
                content = render_search(qs.get("q", [DEFAULT_QUERY])[0].strip() or DEFAULT_QUERY)
            elif parsed.path == "/message":

                content = render_message(mid, qs.get("file", ["0"])[0].strip()) if mid else page("missing message", "<div class='notice bad'>Missing message id.</div>")
            elif parsed.path == "/status":
                content = render_status()
            else:
                self.send_response(404); self.end_headers(); self.wfile.write(b"not found"); return
            data = content.encode("utf-8", "replace")
            self.send_response(200); self.send_header("Content-Type", "text/html; charset=utf-8"); self.send_header("Content-Length", str(len(data))); self.end_headers(); self.wfile.write(data)
        except Exception as e:

## Pilot-safe reindex

This rebuilds only the disposable notmuch database. It must not touch `/mail/Mailstore`.

```sh
set -eu
CONFIG="$HOME/.config/notmuch/default/config"
DB="/mail/SearchIndex/notmuch/default"
BACKUP_ROOT="/mail/Backups/notmuch/reindex-db"
TAG_BACKUP_DIR="/mail/Backups/notmuch"
STAMP="$(date +%Y%m%d-%H%M%S)"
mkdir -p "$BACKUP_ROOT" "$TAG_BACKUP_DIR"

test "$(notmuch --config="$CONFIG" config get database.path)" = "$DB"
test "$(notmuch --config="$CONFIG" config get database.mail_root)" = "/mail/Mailstore"
test "$(notmuch --config="$CONFIG" config get maildir.synchronize_flags)" = "false"
test "$(notmuch --config="$CONFIG" config get index.decrypt)" = "false"

before_msg="$(notmuch --config="$CONFIG" count '*')"
before_files="$(notmuch --config="$CONFIG" count --output=files '*')"
notmuch --config="$CONFIG" search --output=files '*' | LC_ALL=C sort > "$HOME/codex-runs/notmuch-before-$STAMP.paths"

TAG_BACKUP="$TAG_BACKUP_DIR/notmuch-tags-before-reindex-$STAMP.batch-tag"
notmuch --config="$CONFIG" dump --format=batch-tag > "$TAG_BACKUP"
sha256sum "$TAG_BACKUP" > "$TAG_BACKUP.sha256"

DB_BACKUP="$BACKUP_ROOT/default-$STAMP"
mv -- "$DB" "$DB_BACKUP"
mkdir -p "$DB"
notmuch --config="$CONFIG" new
notmuch --config="$CONFIG" restore --format=batch-tag < "$TAG_BACKUP"

after_msg="$(notmuch --config="$CONFIG" count '*')"
after_files="$(notmuch --config="$CONFIG" count --output=files '*')"
notmuch --config="$CONFIG" search --output=files '*' | LC_ALL=C sort > "$HOME/codex-runs/notmuch-after-$STAMP.paths"

echo "before_messages=$before_msg"
echo "before_files=$before_files"
echo "after_messages=$after_msg"
echo "after_files=$after_files"
cmp -s "$HOME/codex-runs/notmuch-before-$STAMP.paths" "$HOME/codex-runs/notmuch-after-$STAMP.paths"
echo "status=reindex_pilot_complete"
echo "db_backup=$DB_BACKUP"
echo "tag_backup=$TAG_BACKUP"
```
