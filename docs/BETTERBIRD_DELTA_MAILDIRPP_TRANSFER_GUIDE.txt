------------

- Close Betterbird and Evolution before conversion, packing, or restore.
- Keep the delta archive separate from the old 59G archive.
- Keep the delta archive separate from mbsync provider-live.
- Never unpack over an existing final destination.
- Verify the archive before transfer, after transfer, and before unpack.
- Run verify-tree and inspect after unpack.
- Add the final directory to Evolution as Maildir-format mail directories.
- Do not choose MH-format mail directories.

from pathlib import Path
from collections import Counter
from email.parser import BytesParser
from email.policy import default
from email.utils import parsedate_to_datetime
import datetime as dt
import os
import sqlite3
import stat

HOME = Path.home()
SRC = HOME / "Betterbird-Email" / "betterbird-maildir"
TARGET = HOME / "Evolution-Mailstore" / "betterbird-archive-maildirpp"
STATE = TARGET / ".conversion-state.sqlite"
LOGROOT = TARGET.parent / "conversion-logs"

def yn(value):
    return "yes" if value else "no"

def fmt_ns(ns):
    return dt.datetime.fromtimestamp(ns / 1_000_000_000).astimezone().isoformat(timespec="seconds")

def scan(root):
    items = []
    if not root.exists():
        return items
    for dirpath, dirnames, filenames in os.walk(root):
        if os.path.basename(dirpath) not in ("cur", "new"):
            continue
        folder = Path(dirpath).parent
        state = Path(dirpath).name
        for name in filenames:
            if name.endswith(".msf"):
                continue
            path = Path(dirpath) / name
            try:
                st = path.stat()
            except FileNotFoundError:
                continue
            if stat.S_ISREG(st.st_mode):
                items.append((path, folder, state, st.st_size, st.st_mtime_ns))
    return items

def date_header(path):
    try:
        data = path.read_bytes()[:1024 * 1024]
        msg = BytesParser(policy=default).parsebytes(data)
        value = msg.get("Date")
        if not value:
            return None, "NO_DATE_HEADER"
        parsed = parsedate_to_datetime(str(value))
        if parsed.tzinfo is None:
            parsed = parsed.replace(tzinfo=dt.timezone.utc)
        return parsed.timestamp(), parsed.astimezone().isoformat(timespec="seconds")
    except Exception as exc:
        return None, "BAD_DATE:" + type(exc).__name__

print("=== Paths ===")
for label, path in (("source", SRC), ("target", TARGET), ("state_db", STATE), ("conversion_log_root", LOGROOT)):
    print(f"{label}={path}")
    print(f"{label}_exists={yn(path.exists())}")
    if path.exists():
        st = path.stat()
        print(f"{label}_mtime={fmt_ns(st.st_mtime_ns)}")
        print(f"{label}_size={st.st_size}")

if LOGROOT.exists():
    dirs = sorted(p for p in LOGROOT.iterdir() if p.is_dir())
    print(f"conversion_log_dir_count={len(dirs)}")
    for p in dirs[:3]:
        print(f"conversion_log_first_candidate={p.name} mtime={fmt_ns(p.stat().st_mtime_ns)}")
    for p in dirs[-3:]:
        print(f"conversion_log_latest_candidate={p.name} mtime={fmt_ns(p.stat().st_mtime_ns)}")

source_items = scan(SRC)
target_items = scan(TARGET)
print("=== Counts ===")
print(f"source_candidate_count={len(source_items)}")
print(f"target_message_count={len(target_items)}")
print(f"source_total_bytes={sum(x[3] for x in source_items)}")
print(f"target_total_bytes={sum(x[3] for x in target_items)}")
print(f"source_cur={sum(1 for x in source_items if x[2] == 'cur')}")
print(f"source_new={sum(1 for x in source_items if x[2] == 'new')}")
print(f"target_cur={sum(1 for x in target_items if x[2] == 'cur')}")
print(f"target_new={sum(1 for x in target_items if x[2] == 'new')}")

state_keys = set()
if STATE.exists():
    con = sqlite3.connect(STATE)
    cur = con.cursor()
    row = cur.execute("select count(*), min(copied_at), max(copied_at) from processed where status='copied'").fetchone()
    print("=== Converter state database ===")
    print(f"state_copied_rows={row[0]}")
    print(f"state_first_copied_at={row[1]}")
    print(f"state_last_copied_at={row[2]}")
    for source_path, size, mtime_ns in cur.execute("select source_path, source_size, source_mtime_ns from processed where status='copied'"):
        state_keys.add((source_path, int(size), int(mtime_ns)))
    con.close()
else:
    print("state_missing=yes")

missing = [x for x in source_items if (str(x[0]), int(x[3]), int(x[4])) not in state_keys]
print("=== Missing candidates ===")
print(f"missing_candidate_count={len(missing)}")
print(f"missing_total_bytes={sum(x[3] for x in missing)}")
for folder, count in Counter(str(x[1].relative_to(SRC)) for x in missing).most_common(20):
    print(f"missing_folder\t{count}\t{folder}")

print("=== Latest missing messages by Date header ===")
dated = []
for path, folder, state, size, mtime_ns in missing:
    ts, text = date_header(path)
    if ts is not None:
        dated.append((ts, text, path, folder, state, size, mtime_ns))
for ts, text, path, folder, state, size, mtime_ns in sorted(dated, reverse=True)[:20]:
    print(f"{text} | state={state} | size={size} | folder={folder.relative_to(SRC)} | path={path.relative_to(SRC)}")

print("=== Batch 1 result ===")
if not SRC.exists() or not TARGET.exists() or not STATE.exists():
    print("status=blocked_missing_required_path")
else:
    print("status=read_only_audit_complete")
PY

from collections import Counter
import hashlib
import os
import shutil
import sqlite3
import stat

HOME = Path.home()
SRC = HOME / "Betterbird-Email" / "betterbird-maildir"

    h = hashlib.sha256()
